
# Ie Chinanbs Monthly


## Metadata

* **Type**: module
* **Language**: python
* **Location**: /dags/ie_chinanbs_monthly.py

## Description

This Python module defines an Apache Airflow DAG for extracting monthly data from the China National Bureau of Statistics (CNBS), transforming it, and loading it into the Insight Engine Postgres database. It scrapes data from the CNBS website using the ChinaNBSDatasource class. The DAG first delays processing until 11:00 AM, then waits for the processed data file to be uploaded to an S3 bucket. It checks if the file exists and either loads it into the database using load_observations_from_csv or skips loading. This allows incrementally loading new monthly data when available while avoiding reprocessing the same data each run.

  from datetime import datetime
  
  import pendulum
  from airflow import DAG
  from airflow.operators.dummy_operator import DummyOperator
  from airflow.operators.python_operator import (BranchPythonOperator,
                                                 PythonOperator,
                                                 ShortCircuitOperator)
  from airflow.providers.amazon.aws.sensors.s3 import S3KeySensor
  
  from insight_engine_etl.chinanbs.scrape import ChinaNBSDatasource
  from insight_engine_etl.common import BUCKET_NAME, get_airflow_default_args
  from insight_engine_etl.database import engine, load_observations_from_csv
  
  release_date = "{{ next_ds }}"
  
  DAG_NAME = 'ie_chinanbs_monthly'
  tz = pendulum.timezone('America/New_York')
  
  default_args = get_airflow_default_args(
      start_date=datetime(2023, 4, 1),
  )
  
  
  dag = DAG(DAG_NAME,
            default_args=default_args,
            catchup=True,
            max_active_runs=1,
            schedule_interval='@monthly')
  
  #cnbs = ChinaNBSDatasource()
  
  
  def exists(observation_file_path, **context):
      if s3.s3_url_exists(observation_file_path):
          return 'load'
      else:
          return 'skip_load_file'
  
  
  def exists(observation_file_path, **context):
      if s3.s3_url_exists(observation_file_path):
          return 'load'
      else:
          return 'skip_load_file'
  
  
  with dag:
      delay = PythonRescheduleOperator(
          task_id='delay_to_11am',
          python_callable=lambda: today(
              hour=11, minute=0, second=0, microsecond=0, tzinfo=tz),
          retries=0
      )
      wait_for_file_task = S3KeySensor(
          task_id="file_sensor_task",
          bucket_name=BUCKET_NAME,
          bucket_key=f'airflow/{DAG_NAME}/{DAG_NAME}__processed__{release_date}.csv.gz',
          wildcard_match=False,
          poke_interval=100,
          timeout=3600 * 1,
          dag=dag
      )
  
      file_exists_task = BranchPythonOperator(
          task_id='check_file_exists',
          python_callable=cnbs.exists,
          # provide_context=True,
          op_kwargs={
              'observation_file_path':  f's3://{BUCKET_NAME}/airflow/{DAG_NAME}/{DAG_NAME}__processed__{release_date}.csv.gz',
              'verbose': True,
          }
      )
  
      load_task = PythonOperator(
          task_id='load',
          python_callable=load_observations_from_csv,
          # provide_context=False,
          op_kwargs={
              'engine': engine,
              'src': f's3://{BUCKET_NAME}/airflow/{DAG_NAME}/{DAG_NAME}__processed__{release_date}.csv.gz',
              'compression': 'gzip'
          }
      )
  
      skip_process_file_task = DummyOperator(task_id='skip_load_file')
  
      wait_for_file_task >> file_exists_task >> [
          load_task, skip_process_file_task]
